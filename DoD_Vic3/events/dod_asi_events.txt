	event_image = { video = "unspecific_politicians_arguing" }

	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		this = c:JAP
		NOT = { exists = c:MCH }
		any_scope_state = {
			count >= 2

		}
		NOT = { has_variable = dod_asi_seen_5 }
	}

	immediate = {
		set_variable = dod_asi_seen_5

		name = dod_asi.5.a
		default_option = yes

			limit = { exists = c:MCH }

		}
		ai_chance = { base = 60 }
	}

	option = {
		name = dod_asi.5.b
		ai_chance = { base = 40 }
	}
}
