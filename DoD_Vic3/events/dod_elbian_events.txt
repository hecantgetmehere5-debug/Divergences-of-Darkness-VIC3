dod_elb.1 = {
	type = country_event
	placement = root
	title = dod_elb.1.t
	desc = dod_elb.1.d
	flavor = dod_elb.1.f
	event_image = { texture = "gfx/event_pictures/dod/wolta_gebeit.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		this = c:CAE
		game_date >= 1850.1.1
		is_at_war = no
		exists = c:ASH
		c:ASH = {
			is_country_type = unrecognized
			is_at_war = no
		}
		NOT = { has_global_variable = dod_fredensborg }
	}

	immediate = {
		set_global_variable = dod_fredensborg
	}

	option = {
		name = dod_elb.1.a
		default_option = yes
		change_infamy = 5
		c:ASH = { trigger_event = { id = dod_elb.2 days = 7 } }
		ai_chance = { base = 75 }
	}

	option = {
		name = dod_elb.1.b
		add_modifier = { name = dod_colonial_humiliation months = 24 }
		ai_chance = { base = 25 }
	}
}
