dod_ita.15 = {
	type = country_event
	placement = root
	title = dod_ita.15.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = dod_ita.15.d_success
				trigger = { var:dod_unification_support >= 10 }
			}
			triggered_desc = {
				desc = dod_ita.15.d_partial
				trigger = { var:dod_unification_support >= 4 }
			}
			triggered_desc = {
				desc = dod_ita.15.d_failure
			}
		}
	}
	flavor = dod_ita.15.f
	event_image = { video = "unspecific_politicians_arguing" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	option = {
		name = dod_ita.15.a
		default_option = yes
		if = {
			limit = { var:dod_unification_support >= 10 }
			every_country = {
				limit = {
					has_variable = dod_council_yes
					dod_is_italian_country = yes
				}
				root = { annex = prev }
			}
			if = {
				limit = { NOT = { exists = c:ITA } }
				change_tag = ITA
			}
			add_modifier = { name = dod_italy_united months = 120 }
			every_country = {
				limit = {
					dod_is_italian_country = yes
					NOT = { this = root }
					NOT = { is_subject_of = root }
				}
				trigger_event = { id = dod_ita.16 days = 14 }
			}
		}
		else_if = {
			limit = { var:dod_unification_support >= 4 }
			every_country = {
				limit = {
					has_variable = dod_council_yes
					dod_is_italian_country = yes
				}
				root = { annex = prev }
			}
		}
		else = {
			add_modifier = { name = dod_council_failed months = 60 }
		}
		every_country = {
			limit = { has_variable = dod_council_yes }
			remove_variable = dod_council_yes
		}
		remove_global_variable = dod_council_host
	}
}
