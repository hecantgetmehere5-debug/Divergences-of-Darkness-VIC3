namespace = dod_rail

dod_rail.1 = {
	type = country_event
	placement = root
	title = dod_rail.1.t
	desc = dod_rail.1.d
	flavor = dod_rail.1.f
	event_image = { video = "southamerica_war_civilians" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		this = c:SPA
		exists = c:CLM
		game_date < 1837.1.1
		is_at_war = no
		NOT = { has_global_variable = dod_rail_spain }
	}

	immediate = {
		set_global_variable = dod_rail_spain
	}

	option = {
		name = dod_rail.1.a
		default_option = yes
		create_diplomatic_play = {
			target_country = c:CLM
			type = dp_annex_war
		}
		ai_chance = { base = 90 }
	}

	option = {
		name = dod_rail.1.b
		ai_chance = { base = 10 }
	}
}

dod_rail.10 = {
	type = country_event
	placement = root
	title = dod_rail.10.t
	desc = dod_rail.10.d
	flavor = dod_rail.10.f
	event_image = { video = "unspecific_ruler_speaking_to_people" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		this = c:HLR
		game_date >= 1837.1.1
		has_journal_entry = je_dod_prague_proclamation
		NOT = { has_variable = dod_rail_imperial_lands }
	}

	immediate = {
		set_variable = dod_rail_imperial_lands
	}

	option = {
		name = dod_rail.10.a
		default_option = yes
		change_infamy = 5
		every_state = {
			limit = {
				state_region = {
					OR = {
						is_homeland = cu:north_german
						is_homeland = cu:rheinish
					}
				}
				owner = {
					OR = {
						this = c:SCA
						this = c:BRG
					}
				}
			}
			state_region = { add_claim = root }
		}
		ai_chance = { base = 80 }
	}

	option = {
		name = dod_rail.10.b
		ai_chance = { base = 20 }
	}
}
