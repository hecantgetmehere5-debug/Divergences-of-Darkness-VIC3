dod_dua.10 = {
	type = country_event
	placement = root
	title = dod_dua.10.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = dod_dua.10.d_elisabeth
				trigger = { var:dod_succession = 1 }
			}
			triggered_desc = {
				desc = dod_dua.10.d_edouard
				trigger = { var:dod_succession = 2 }
			}
			triggered_desc = {
				desc = dod_dua.10.d_gascony
				trigger = { var:dod_succession = 3 }
			}
			triggered_desc = {
				desc = dod_dua.10.d_natural
			}
		}
	}
	flavor = dod_dua.10.f
	event_image = { texture = "gfx/event_pictures/dod/Death_King.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	trigger = {
		c:DUA ?= this
		NOT = { has_variable = dod_succession_done }
		OR = {
			AND = {
				has_variable = dod_heir_killed
				game_date >= 1845.1.1
			}
			game_date >= 1861.1.1
		}
	}

	immediate = {
		set_variable = dod_succession_done
		if = {
			limit = { has_variable = dod_heir_killed }
			random_list = {
				45 = { set_variable = { name = dod_succession value = 1 } }
				30 = { set_variable = { name = dod_succession value = 2 } }
				25 = { set_variable = { name = dod_succession value = 3 } }
			}
		}
		else = {
			set_variable = { name = dod_succession value = 4 }
		}
	}

	option = {
		name = dod_dua.10.a
		default_option = yes
		if = {
			limit = { var:dod_succession = 1 }
			create_character = { first_name = "Elisabeth" last_name = "Plantagenet" female = yes ruler = yes age = 28 }
			dod_dua_shift = { ENGLISH = -5 FRENCH = 5 ACCEPT = 0 }
		}
		else_if = {
			limit = { var:dod_succession = 2 }
			create_character = { first_name = "Edouard" last_name = "Plantagenet" ruler = yes age = 40 }
			dod_dua_shift = { ENGLISH = -15 FRENCH = 15 ACCEPT = 0 }
		}
		else_if = {
			limit = { var:dod_succession = 3 }
			create_character = { first_name = "Henri" last_name = "Plantagenet" ruler = yes age = 35 }
			dod_dua_shift = { ENGLISH = 15 FRENCH = -15 ACCEPT = 0 }
		}
		else = {
			create_character = { first_name = "Henri" last_name = "Plantagenet" ruler = yes age = 45 }
			dod_dua_shift = { ENGLISH = -5 FRENCH = -5 ACCEPT = 5 }
		}
		remove_variable = dod_succession
	}
}
