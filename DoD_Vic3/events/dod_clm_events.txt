dod_clm.15 = {
	type = country_event
	placement = root
	title = dod_clm.15.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = dod_clm.15.d_united
				trigger = { var:dod_colombian_unity >= 3 }
			}
			triggered_desc = {
				desc = dod_clm.15.d_partial
				trigger = { var:dod_colombian_unity >= 0 }
			}
			triggered_desc = {
				desc = dod_clm.15.d_failure
			}
		}
	}
	flavor = dod_clm.15.f
	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	option = {
		name = dod_clm.15.a
		default_option = yes
		if = {
			limit = { var:dod_colombian_unity >= 3 }
			set_variable = dod_clm_united
			add_modifier = { name = dod_colombian_unity months = 120 }
		}
		else_if = {
			limit = { var:dod_colombian_unity >= 0 }
			dod_release_homeland = { TAG = UCA CULTURE = central_american }
			add_modifier = { name = dod_colombia_divided months = 60 }
		}
		else = {
			dod_release_homeland = { TAG = UCA CULTURE = central_american }
			dod_release_homeland = { TAG = GRA CULTURE = granadan }
			dod_release_homeland = { TAG = LST CULTURE = lusitanian }
			change_tag = MEX
			add_modifier = { name = dod_colombia_divided months = 120 }
		}
	}
}
