# The diets report back
dod_hre.20 = {
	type = country_event
	placement = root
	title = dod_hre.20.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = dod_hre.20.d_failure
				trigger = { var:dod_federations >= 3 }
			}
			triggered_desc = {
				desc = dod_hre.20.d_success
				trigger = { var:dod_federations = 0 }
			}
			triggered_desc = {
				desc = dod_hre.20.d_mixed
			}
		}
	}
	flavor = dod_hre.20.f
	event_image = { video = "europenorthamerica_before_the_battle" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	immediate = {
		if = {
			limit = {
				var:dod_elbe_refuse >= 2
				var:dod_elbe_refuse > var:dod_elbe_accept
			}
			dod_hre_form_federation = { REGION = elbian TAG = NGF }
		}
		if = {
			limit = {
				var:dod_rhine_refuse >= 2
				var:dod_rhine_refuse > var:dod_rhine_accept
			}
			dod_hre_form_federation = { REGION = rhenish TAG = RHI }
		}
		if = {
			limit = {
				var:dod_danube_refuse >= 2
				var:dod_danube_refuse > var:dod_danube_accept
			}
			dod_hre_form_federation = { REGION = danubian TAG = SGF }
		}
	}

	# Submission or conquest
	option = {
		name = dod_hre.20.a
		default_option = yes
		every_country = {
			limit = { has_variable = dod_rebel_federation }
			save_scope_as = dod_federation
			root = {
				create_diplomatic_play = {
					target_country = scope:dod_federation
					type = dp_annex_war
				}
			}
		}
		add_journal_entry = { type = je_dod_submission_or_conquest }
		ai_chance = { base = 70 }
	}

	# Give the rebels time
	option = {
		name = dod_hre.20.b
		trigger = { var:dod_federations > 0 }
		add_journal_entry = { type = je_dod_submission_or_conquest }
		ai_chance = { base = 30 }
	}
}

# Every prince has submitted: what shape does the Reich take?
dod_hre.40 = {
	type = country_event
	placement = root
	title = dod_hre.40.t
	desc = dod_hre.40.d
	flavor = dod_hre.40.f
	event_image = { video = "ip4_abdication" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	# Ein Reich: one German state, ruled from Frankfurt
	option = {
		name = dod_hre.40.a
		default_option = yes
		every_country = {
			limit = {
				dod_is_imperial_prince = yes
				is_subject_of = root
			}
			root = { annex = prev }
		}
		add_primary_culture = cu:north_german
		add_primary_culture = cu:rheinish
		add_primary_culture = cu:south_german
		add_modifier = { name = dod_imperial_hegemony months = 120 }
		if = {
			limit = {
				country_has_primary_culture = cu:czech
				any_scope_state = { dod_is_bohemian_state = yes }
			}
			remove_primary_culture = cu:czech
			trigger_event = { id = dod_hre.50 days = 30 }
		}
		ai_chance = { base = 50 }
	}

	# One Reich of two crowns: Bohemia keeps its place
	option = {
		name = dod_hre.40.b
		every_country = {
			limit = {
				dod_is_imperial_prince = yes
				is_subject_of = root
			}
			root = { annex = prev }
		}
		add_primary_culture = cu:north_german
		add_primary_culture = cu:rheinish
		add_primary_culture = cu:south_german
		add_modifier = { name = dod_imperial_hegemony months = 60 }
		ai_chance = { base = 30 }
	}

	# Keep the princes as vassals of a federal Reich
	option = {
		name = dod_hre.40.c
		add_modifier = { name = dod_federal_reich months = 240 }
		ai_chance = { base = 20 }
	}
}

# The Bohemian Revolution
dod_hre.54 = {
	type = country_event
	placement = root
	title = dod_hre.54.t
	desc = dod_hre.54.d
	flavor = dod_hre.54.f
	event_image = { video = "europenorthamerica_before_the_battle" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		NOT = { exists = c:BOH }
		any_scope_state = { dod_is_bohemian_state = yes }
	}

	immediate = {
		random_scope_state = {
			limit = { dod_is_bohemian_state = yes }
			save_scope_as = dod_bohemian_heartland
		}
		create_country = {
			tag = BOH
			origin = root
			state = scope:dod_bohemian_heartland
		}
		every_scope_state = {
			limit = { dod_is_bohemian_state = yes }
			set_state_owner = c:BOH
		}
	}

	# The Empire strikes back
	option = {
		name = dod_hre.54.a
		default_option = yes
		create_diplomatic_play = {
			target_country = c:BOH
			type = dp_annex_war
		}
		add_journal_entry = { type = je_dod_war_for_bohemia }
		c:BOH = { trigger_event = { id = dod_hre.57 } }
		every_country = {
			limit = {
				OR = {
					c:SCA ?= this
					c:BRG ?= this

			}
			trigger_event = { id = dod_hre.55 days = 14 }

	option = {
		name = dod_hre.54.b

}
