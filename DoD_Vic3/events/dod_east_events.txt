namespace = dod_east

dod_east.1 = {
	type = country_event
	placement = root
	title = dod_east.1.t
	desc = dod_east.1.d
	flavor = dod_east.1.f
	event_image = { video = "ep2_important_japanese_men_negotiating" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		c:TNG ?= this
		exists = c:MIG
		is_at_war = no
		game_date >= 1840.1.1
		NOT = { has_variable = dod_east_seen_1 }
	}

	immediate = {
		set_variable = dod_east_seen_1
	}

	option = {
		name = dod_east.1.a
		default_option = yes
		create_diplomatic_play = {
			target_country = c:MIG
			type = dp_annex_war
		}
		set_variable = dod_regency_war
		ai_chance = { base = 70 }
	}

	option = {
		name = dod_east.1.b
		add_modifier = { name = dod_self_strengthening months = 60 }
		ai_chance = { base = 30 }
	}
}

dod_east.10 = {
	type = country_event
	placement = root
	title = dod_east.10.t
	desc = dod_east.10.d
	flavor = dod_east.10.f
	event_image = { video = "ep2_important_japanese_men_negotiating" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		c:JAP ?= this
		game_date >= 1855.1.1
		is_at_war = no
		NOT = { exists = c:KOR }
		NOT = { has_modifier = dod_korean_autonomy }
		any_scope_state = { state_region = { is_homeland = cu:korean } }
		NOT = { has_variable = dod_east_seen_10 }
	}

	immediate = {
		set_variable = dod_east_seen_10
		dod_release_homeland = { TAG = KOR CULTURE = korean }
	}

	option = {
		name = dod_east.10.a
		default_option = yes
		if = {
			limit = { exists = c:KOR }
			create_diplomatic_play = {
				target_country = c:KOR
				type = dp_annex_war
			}
		}
		ai_chance = { base = 70 }
	}

	option = {
		name = dod_east.10.b
		add_modifier = { name = dod_enforced_peace months = 60 }
		ai_chance = { base = 30 }
	}
}
