namespace = dod_afr

dod_afr.1 = {
	type = country_event
	placement = root
	title = dod_afr.1.t
	desc = dod_afr.1.d
	flavor = dod_afr.1.f
	event_image = { texture = "gfx/event_pictures/dod/boer_war.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		c:BRG ?= this
		is_at_war = no
		any_country = {
			country_has_primary_culture = cu:boer
			NOT = { is_subject_of = c:BRG }
		}
		NOT = { has_variable = dod_afr_seen_1 }
	}

	immediate = {
		set_variable = dod_afr_seen_1
	}

	option = {
		name = dod_afr.1.a
		default_option = yes
		every_country = {
			limit = {
				country_has_primary_culture = cu:boer
				NOT = { is_subject_of = root }
				country_rank < rank_value:major_power
			}
			trigger_event = { id = dod_afr.2 days = 14 }
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = dod_afr.1.b
		ai_chance = { base = 40 }
	}
}
