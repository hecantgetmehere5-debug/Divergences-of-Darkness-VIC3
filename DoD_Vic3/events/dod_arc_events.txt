dod_arc.1 = {
	type = country_event
	placement = root
	title = dod_arc.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = dod_arc.1.d_main
			}
		}
	}
	flavor = dod_arc.1.f
	event_image = { video = "middleeast_jungle_expedition" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	trigger = {
		has_variable = dod_dig_requester
	}

	option = {
		name = dod_arc.1.a
		default_option = yes
		var:dod_dig_requester = {
			add_journal_entry = { type = je_dod_excavation }
		}
		change_relations = { country = var:dod_dig_requester value = 20 }
		remove_variable = dod_dig_requester
		ai_chance = { base = 60 }
	}

	option = {
		name = dod_arc.1.b
		var:dod_dig_requester = { trigger_event = { id = dod_arc.2 days = 3 } }
		remove_variable = dod_dig_requester
		ai_chance = { base = 40 }
	}
}

				NOT = { has_variable = dod_discoveries }
			}
			trigger_event = { id = dod_arc.22 days = 14 }
		}
	}
}
