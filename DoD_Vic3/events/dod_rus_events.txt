dod_rus.1 = {
	type = country_event
	placement = root
	title = dod_rus.1.t
	desc = dod_rus.1.d
	flavor = dod_rus.1.f
	event_image = { video = "votp_garibaldi" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	trigger = {
		exists = c:RUS
		country_has_primary_culture = cu:russian
		NOT = { this = c:RUS }
	}

	option = {
		name = dod_rus.1.a
		default_option = yes
		c:RUS = { annex = root }
		ai_chance = { base = 50 }
	}

	option = {
		name = dod_rus.1.b

			set_variable = { name = dod_russian_holdout value = root }
			trigger_event = { id = dod_rus.6 days = 3 }
		}
		ai_chance = { base = 50 }
	}
}
