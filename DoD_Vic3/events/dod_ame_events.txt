namespace = dod_ame

dod_ame.1 = {
	type = country_event
	placement = root
	title = dod_ame.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = dod_ame.1.d_vinlander
				trigger = { var:dod_conference_people = 1 }
			}
			triggered_desc = {
				desc = dod_ame.1.d_septiman
				trigger = { var:dod_conference_people = 2 }
			}
			triggered_desc = {
				desc = dod_ame.1.d_plantagenian
				trigger = { var:dod_conference_people = 3 }
			}
			triggered_desc = {
				desc = dod_ame.1.d_lusitanian
			}
		}
	}
	flavor = dod_ame.1.f
	event_image = { video = "europenorthamerica_rich_and_poor" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	immediate = {
		# Wiki: Arcadianism is strongest among the Septimans and weakest in Vinland,
		# still bitter after the Treaty of Lofby.
		if = {
			limit = { var:dod_conference_people = 1 }
			random_list = {
				25 = { set_variable = { name = dod_conference_result value = 1 } }
				35 = { set_variable = { name = dod_conference_result value = 2 } }
				40 = { set_variable = { name = dod_conference_result value = 3 } }
			}
		}
		else_if = {
			limit = { var:dod_conference_people = 2 }
			random_list = {
				70 = { set_variable = { name = dod_conference_result value = 1 } }
				25 = { set_variable = { name = dod_conference_result value = 2 } }
				5 = { set_variable = { name = dod_conference_result value = 3 } }
			}
		}
		else = {
			random_list = {
				50 = { set_variable = { name = dod_conference_result value = 1 } }
				30 = { set_variable = { name = dod_conference_result value = 2 } }
				20 = { set_variable = { name = dod_conference_result value = 3 } }
			}
		}
	}

	option = {
		name = dod_ame.1.a
		trigger = { var:dod_conference_result = 1 }
		default_option = yes
		if = {
			limit = { var:dod_conference_people = 1 }
			add_primary_culture = cu:vinlander
		}
		if = {
			limit = { var:dod_conference_people = 2 }
			add_primary_culture = cu:septiman
		}
		if = {
			limit = { var:dod_conference_people = 3 }
			add_primary_culture = cu:plantagenian
		}
		if = {
			limit = { var:dod_conference_people = 4 }
			add_primary_culture = cu:lusitanian
		}
		change_variable = { name = dod_arcadian_accepted add = 1 }
	}

	option = {
		name = dod_ame.1.b
		trigger = { var:dod_conference_result = 2 }
		if = {
			limit = { var:dod_conference_people = 1 }
			add_primary_culture = cu:vinlander
		}
		if = {
			limit = { var:dod_conference_people = 2 }
			add_primary_culture = cu:septiman
		}
		if = {
			limit = { var:dod_conference_people = 3 }
			add_primary_culture = cu:plantagenian
		}
		if = {
			limit = { var:dod_conference_people = 4 }
			add_primary_culture = cu:lusitanian
		}
		change_variable = { name = dod_arcadian_accepted add = 1 }
		add_modifier = { name = dod_arcadian_concessions months = 60 }
	}

	option = {
		name = dod_ame.1.c
		trigger = { var:dod_conference_result = 2 }
		add_modifier = { name = dod_arcadian_separatism months = 60 }
	}

	option = {
		name = dod_ame.1.opt_d
		trigger = { var:dod_conference_result = 3 }
		add_modifier = { name = dod_arcadian_separatism months = 120 }
	}
}

	}

	option = {
		name = dod_ame.10.a

		if = {
			limit = { var:dod_purchase_kind = 1 }
			every_scope_state = {
				limit = { state_region = { is_homeland = cu:athesian } }

		else_if = {

		}
		else = {
			every_scope_state = {
				limit = { state_region = { is_homeland = cu:ayitian } }
				set_state_owner = var:dod_purchase_buyer
			}
		}
		var:dod_purchase_buyer = { add_modifier = { name = dod_colonial_purchase months = 60 } }
		remove_variable = dod_purchase_buyer
		remove_variable = dod_purchase_kind
