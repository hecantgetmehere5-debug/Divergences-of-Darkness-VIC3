# Hungary storyline: decisions, ported from the Vic2 mod's decisions/Hungary.txt.
# The reforms (dod_hun_reform_*) each count towards je_dod_hungarian_centralisation.

dod_hun_abolish_szekely_privilege = {
	is_shown = {
		c:HUN ?= this
		has_journal_entry = je_dod_hungarian_centralisation
		NOT = { has_variable = dod_hun_szekely_abolished }
	}
	possible = { is_at_war = no }
	when_taken = {
		set_variable = dod_hun_szekely_abolished
		change_variable = { name = dod_hun_reforms add = 1 }
		trigger_event = { id = dod_hun.10 days = 30 }
		trigger_event = { id = dod_hun.12 days = 365 }
	}
	ai_chance = { value = 50 }
}

dod_hun_reclaim_croatia = {
	is_shown = {
		c:HUN ?= this
		has_variable = dod_hun_crowned
		exists = c:CRO
		NOT = { owns_entire_state_region = STATE_CROATIA }
		NOT = { has_variable = dod_hun_croatia_claimed }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:major_power
	}
	when_taken = {
		set_variable = dod_hun_croatia_claimed
		s:STATE_CROATIA = { add_claim = root }
		s:STATE_SLAVONIA = { add_claim = root }
		s:STATE_BOSNIA = { add_claim = root }
		change_relations = { country = c:CRO value = -30 }
	}
	ai_chance = { value = 30 }
}
