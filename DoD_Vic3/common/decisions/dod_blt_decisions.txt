# Baltic, Scandinavia & Poland-Lithuania

dod_abolish_courland = {
	is_shown = {
		c:SCA ?= this
		exists = c:KRL
		c:KRL = { is_subject_of = root }
	}
	possible = {
		is_at_war = no
	}
	when_taken = {
		annex = c:KRL
	}
	ai_chance = { value = 60 }
}

dod_annex_kurland = {
	is_shown = {
		c:SCA ?= this
		exists = c:KRL
		c:KRL = { NOT = { is_subject_of = root } }
		NOT = { has_variable = dod_kurland_war }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:great_power
	}
	when_taken = {
		set_variable = dod_kurland_war
		create_diplomatic_play = {
			target_country = c:KRL
			type = dp_annex_war
		}
	}
	ai_chance = { value = 30 }
}

dod_dominium_maris_baltici = {
	is_shown = {
		c:SCA ?= this
		NOT = { has_variable = dod_dominium }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:great_power
	}
	when_taken = {
		set_variable = dod_dominium
		change_infamy = 15
		add_modifier = { name = dod_baltic_dominion months = 120 }
		every_state = {
			limit = {
				state_region = {
					OR = {
					is_homeland = cu:latvian
					is_homeland = cu:estonian
					is_homeland = cu:livlandian
					}
				}
				NOT = { owner = root }
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { value = 50 }
}

			}

		}
	}
	possible = {
		is_at_war = no
	}

		}

	}
	ai_chance = { value = 60 }
}

dod_reform_grand_duchy = {
	is_shown = {
		c:LIT ?= this
		NOT = { has_variable = dod_grand_duchy }
	}
	possible = {
		is_at_war = no
		OR = {
			NOT = { exists = c:PLC }
			c:PLC = { country_rank < rank_value:great_power }
		}
		any_scope_state = { state_region = { is_homeland = cu:byelorussian } }
	}
	when_taken = {
		set_variable = dod_grand_duchy
		add_primary_culture = cu:byelorussian
		add_modifier = { name = dod_grand_duchy_reformed months = 120 }
		every_state = {
			limit = {
				state_region = {
					OR = {
					is_homeland = cu:lithuanian
					is_homeland = cu:byelorussian
					}
				}
				NOT = { owner = root }
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { value = 80 }
}
