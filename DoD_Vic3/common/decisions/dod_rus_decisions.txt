# Russia, the Steppe & the Caucasus

dod_form_russia = {
	is_shown = {
		country_has_primary_culture = cu:russian
		NOT = { exists = c:RUS }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 6
			state_region = {
				is_homeland = cu:russian
			}
		}
	}
	when_taken = {
		change_tag = RUS
		every_country = {
			limit = {
				country_has_primary_culture = cu:russian
				NOT = { this = root }
			}
			trigger_event = { id = dod_rus.1 days = 7 }
		}
	}
	ai_chance = { base = 100 }
}

dod_rus_integrate = {
	is_shown = {
		this = c:RUS
		NOT = { has_variable = dod_rus_integrated }
	}
	possible = {
		is_at_war = no
		game_date >= 1850.1.1
	}
	when_taken = {
		set_variable = dod_rus_integrated
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:ruthenian } }
			}
			add_primary_culture = cu:ruthenian
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:tatar } }
			}
			add_primary_culture = cu:tatar
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:astrakhani } }
			}
			add_primary_culture = cu:astrakhani
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:crimean } }
			}
			add_primary_culture = cu:crimean
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:georgian } }
			}
			add_primary_culture = cu:georgian
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:estonian } }
			}
			add_primary_culture = cu:estonian
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:latvian } }
			}
			add_primary_culture = cu:latvian
		}
		if = {
			limit = {
				any_scope_state = { state_region = { is_homeland = cu:finnish } }
			}
			add_primary_culture = cu:finnish
		}
	}
	ai_chance = { base = 60 }
}

dod_form_tatarstan = {
	is_shown = {
		OR = {
			country_has_primary_culture = cu:crimean
			country_has_primary_culture = cu:astrakhani
			country_has_primary_culture = cu:tatar
		}
		NOT = { exists = c:TAR }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 4
			state_region = {
				OR = {
					is_homeland = cu:crimean
					is_homeland = cu:astrakhani
					is_homeland = cu:tatar
				}
			}
		}
	}
	when_taken = {
		every_country = {
			limit = {
				is_subject_of = root
				OR = {
					country_has_primary_culture = cu:crimean
					country_has_primary_culture = cu:astrakhani
					country_has_primary_culture = cu:tatar
				}
			}
			root = { annex = prev }
		}
		add_primary_culture = cu:crimean
		add_primary_culture = cu:astrakhani
		add_primary_culture = cu:tatar
		change_tag = TAR
	}
	ai_chance = { base = 100 }
}
