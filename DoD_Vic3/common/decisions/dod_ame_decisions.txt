# Arcadia & Ameriga

dod_form_arcadia = {
	is_shown = {
		OR = {
			c:VIN ?= this
			c:NBG ?= this
			c:PLA ?= this
		}
		NOT = { exists = c:ARC }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:major_power
		any_scope_state = {
			count >= 2
			state_region = {
				OR = {
					is_homeland = cu:vinlander
					is_homeland = cu:septiman
					is_homeland = cu:plantagenian
				}
			}
		}
	}
	when_taken = {
		change_tag = ARC
		add_journal_entry = { type = je_dod_arcadian_union }
	}
	ai_chance = { value = 100 }
}

dod_arcadian_conference = {
	is_shown = {
		c:ARC ?= this
		has_journal_entry = je_dod_arcadian_union
	}
	possible = {
		OR = {
			AND = {
				NOT = { has_variable = dod_conference_1 }
				NOT = { country_has_primary_culture = cu:vinlander }
				any_scope_state = { state_region = { is_homeland = cu:vinlander } }
			}
			AND = {
				NOT = { has_variable = dod_conference_2 }
				NOT = { country_has_primary_culture = cu:septiman }
				any_scope_state = { state_region = { is_homeland = cu:septiman } }
			}
			AND = {
				NOT = { has_variable = dod_conference_3 }
				NOT = { country_has_primary_culture = cu:plantagenian }
				any_scope_state = { state_region = { is_homeland = cu:plantagenian } }
			}
			AND = {
				NOT = { has_variable = dod_conference_4 }
				NOT = { country_has_primary_culture = cu:lusitanian }
				any_scope_state = { state_region = { is_homeland = cu:lusitanian } }
			}
		}
		is_at_war = no
	}
	when_taken = {
		if = {
			limit = {
				NOT = { has_variable = dod_conference_1 }
				NOT = { country_has_primary_culture = cu:vinlander }
				any_scope_state = { state_region = { is_homeland = cu:vinlander } }
			}
			set_variable = dod_conference_1
			set_variable = { name = dod_conference_people value = 1 }
		}
		else_if = {
			limit = {
				NOT = { has_variable = dod_conference_2 }
				NOT = { country_has_primary_culture = cu:septiman }
				any_scope_state = { state_region = { is_homeland = cu:septiman } }
			}
			set_variable = dod_conference_2
			set_variable = { name = dod_conference_people value = 2 }
		}
		else_if = {
			limit = {
				NOT = { has_variable = dod_conference_3 }
				NOT = { country_has_primary_culture = cu:plantagenian }
				any_scope_state = { state_region = { is_homeland = cu:plantagenian } }
			}
			set_variable = dod_conference_3
			set_variable = { name = dod_conference_people value = 3 }
		}
		else_if = {
			limit = {
				NOT = { has_variable = dod_conference_4 }
				NOT = { country_has_primary_culture = cu:lusitanian }
				any_scope_state = { state_region = { is_homeland = cu:lusitanian } }
			}
			set_variable = dod_conference_4
			set_variable = { name = dod_conference_people value = 4 }
		}
		trigger_event = { id = dod_ame.1 days = 30 }
	}
	ai_chance = { value = 80 }
}

dod_arcadia_integrate = {
	is_shown = {
		OR = {
			c:ARC ?= this
			c:VIN ?= this
			c:NBG ?= this
			c:PLA ?= this
		}
		NOT = { has_variable = dod_arcadia_integrated }
	}
	possible = {
		is_at_war = no
		game_date >= 1850.1.1
	}
	when_taken = {
		set_variable = dod_arcadia_integrated
		if = {
			limit = { any_scope_state = { state_region = { is_homeland = cu:athesian } } }
			add_primary_culture = cu:athesian
		}
		if = {
			limit = { any_scope_state = { state_region = { is_homeland = cu:caribeno } } }
			add_primary_culture = cu:caribeno
		}
		if = {
			limit = { any_scope_state = { state_region = { is_homeland = cu:ayitian } } }
			add_primary_culture = cu:ayitian
		}
		if = {
			limit = { any_scope_state = { state_region = { is_homeland = cu:afro_germanic } } }
			add_primary_culture = cu:afro_germanic
		}
		if = {
			limit = { any_scope_state = { state_region = { is_homeland = cu:rohnnst } } }
			add_primary_culture = cu:rohnnst
		}
	}
	ai_chance = { value = 70 }
}

dod_purchase_athesia = {
	is_shown = {
		OR = {
			c:ARC ?= this
			c:PLA ?= this
			c:NBG ?= this
			c:VIN ?= this
		}
		NOT = { has_variable = dod_purchase_athesia }
		any_state = {
			state_region = { is_homeland = cu:athesian }
			NOT = { owner = root }
		}
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:major_power
	}
	when_taken = {
		set_variable = dod_purchase_athesia
		random_state = {
			limit = {
				state_region = { is_homeland = cu:athesian }
				NOT = { owner = root }
			}
			owner = {
				set_variable = { name = dod_purchase_buyer value = root }
				set_variable = { name = dod_purchase_kind value = 1 }
				trigger_event = { id = dod_ame.10 days = 7 }
			}
		}
	}
	ai_chance = { value = 40 }
}

dod_form_ameriga = {
	is_shown = {
		OR = {
			c:AMA ?= this
			c:LOT ?= this
		}
		NOT = { exists = c:AZL }
	}
	possible = {
		is_at_war = no
		any_scope_state = { state_region = { is_homeland = cu:amazonian } }
		any_scope_state = { state_region = { is_homeland = cu:lotharian } }
	}
	when_taken = {
		every_country = {
			limit = {
				is_subject_of = root
				OR = {
					country_has_primary_culture = cu:amazonian
					country_has_primary_culture = cu:lotharian
				}
			}
			root = { annex = prev }
		}
		add_primary_culture = cu:amazonian
		add_primary_culture = cu:lotharian
		change_tag = AZL
		add_modifier = { name = dod_ameriga_federation months = 120 }
		trigger_event = { id = dod_ame.21 days = 30 }
	}
	ai_chance = { value = 100 }
}
