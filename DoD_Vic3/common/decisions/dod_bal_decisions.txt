dod_tanzimat_1 = {

	}

}

dod_reclaim_macedonia = {
	is_shown = {
		c:TUR ?= this
		NOT = { has_variable = dod_reclaimed_macedonia }
	}
	possible = {
		is_at_war = no
		NOT = { has_journal_entry = je_dod_ottoman_civil_war }
		country_rank >= rank_value:major_power
	}
	when_taken = {
		set_variable = dod_reclaimed_macedonia
		change_infamy = 10
		every_state = {
			limit = {
				state_region = {
					OR = {
						is_homeland = cu:greek
						is_homeland = cu:bulgarian
					}
				}
				NOT = { owner = root }
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { value = 30 }
}

dod_form_yugoslavia = {
	is_shown = {
		OR = {
			country_has_primary_culture = cu:serb
			country_has_primary_culture = cu:croat
			country_has_primary_culture = cu:slovene
			country_has_primary_culture = cu:bosniak
		}
		NOT = { exists = c:YUG }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 5
			state_region = {
				OR = {
					is_homeland = cu:serb
					is_homeland = cu:croat
					is_homeland = cu:slovene
					is_homeland = cu:bosniak
				}
			}
		}
	}
	when_taken = {
		add_primary_culture = cu:serb
		add_primary_culture = cu:croat
		add_primary_culture = cu:slovene
		add_primary_culture = cu:bosniak
		change_tag = YUG
	}
	ai_chance = { value = 100 }
}
