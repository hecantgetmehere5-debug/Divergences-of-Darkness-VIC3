# Italy, Aragon-Italy, the Valois & Venice-Greece

dod_call_council_of_italy = {
	is_shown = {
		OR = {
			c:TUS ?= this
			c:VEN ?= this
		}
		NOT = { exists = c:ITA }
		NOT = { has_global_variable = dod_council_host }
	}
	possible = {
		is_at_war = no
		dod_is_republic = yes
		NOT = { has_modifier = dod_council_failed }
		trigger_if = {
			limit = { c:VEN ?= this }
			game_date >= 1860.1.1
		}
	}
	when_taken = {
		add_journal_entry = { type = je_dod_council_of_italy }
	}
	ai_chance = { value = 50 }
}

dod_proclaim_aragon_italy = {
	is_shown = {
		c:ARA ?= this
		NOT = { exists = c:ARI }
		NOT = { exists = c:ITA }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 2
			dod_is_north_italian_state = yes
		}
		any_scope_state = {
			count >= 2
			dod_is_south_italian_state = yes
		}
	}
	when_taken = {
		add_primary_culture = cu:south_italian
		change_tag = ARI
		every_country = {
			limit = {
				country_rank >= rank_value:great_power
				NOT = { this = root }
			}
			trigger_event = { id = dod_ita.40 days = 7 }
		}
	}
	ai_chance = { value = 100 }
}

dod_valois_restore_italy = {
	is_shown = {
		c:PRV ?= this
		NOT = { exists = c:ITA }
		NOT = { exists = c:ARI }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 3
			dod_is_north_italian_state = yes
		}
	}
	when_taken = {
		set_global_variable = dod_valois_italy
		add_primary_culture = cu:north_italian
		add_primary_culture = cu:south_italian
		change_tag = ITA
	}
	ai_chance = { value = 100 }
}

dod_valois_challenge_usurpers = {
	is_shown = {
		OR = {
			c:PRV ?= this
			AND = {
				c:ITA ?= this
				has_global_variable = dod_valois_italy
			}
		}
		NOT = { has_variable = dod_valois_claims }
	}
	possible = {
		is_at_war = no
		has_law = law_type:law_monarchy
		any_scope_state = { state_region = { is_homeland = cu:provencal } }
	}
	when_taken = {
		set_variable = dod_valois_claims
		change_infamy = 15
		every_state = {
			limit = {
				OR = {
					dod_is_french_state = yes
					state_region = { is_homeland = cu:provencal }
				}
				NOT = { owner = root }
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { value = 50 }
}
