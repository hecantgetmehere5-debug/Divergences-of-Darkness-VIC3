# Persia & the Middle East

dod_unite_iran = {
	is_shown = {
		country_has_primary_culture = cu:persian
		NOT = { exists = c:PER }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 6
			state_region = {
				is_homeland = cu:persian
			}
		}
	}
	when_taken = {
		every_country = {
			limit = {
				is_subject_of = root
				country_has_primary_culture = cu:persian
			}
			root = { annex = prev }
		}
		change_tag = PER
	}
	ai_chance = { base = 100 }
}

dod_iranshahr = {
	is_shown = {
		this = c:PER
		NOT = { has_variable = dod_iranshahr }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:major_power
	}
	when_taken = {
		set_variable = dod_iranshahr
		change_infamy = 15
		every_state = {
			limit = {
				state_region = {
					OR = {
						is_homeland = cu:persian
						is_homeland = cu:lurish
						is_homeland = cu:baluchi
						is_homeland = cu:azerbaijani
						is_homeland = cu:kurdish
					}
				}
				NOT = { owner = root }
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { base = 50 }
}
