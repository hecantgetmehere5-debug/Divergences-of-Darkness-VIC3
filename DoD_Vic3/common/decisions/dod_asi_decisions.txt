# East, Southeast & South Asia

dod_seoul_treaty = {
	is_shown = {
		this = c:JAP
		exists = c:KOR
		NOT = { has_variable = dod_seoul_treaty }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:major_power
		c:KOR = { NOT = { is_subject_of = root } }
	}
	when_taken = {
		set_variable = dod_seoul_treaty
		c:KOR = { trigger_event = { id = dod_asi.1 days = 7 } }
	}
	ai_chance = { base = 60 }
}

dod_unequal_treaty_decision = {
	is_shown = {
		country_rank >= rank_value:great_power
		NOT = { OR = { 	this = c:CHI 	this = c:MIG 	this = c:TNG } }
		NOT = { has_variable = dod_unequal_treaty_sent }
	}
	possible = {
		is_at_war = no
		any_country = {
			OR = {
				this = c:CHI
				this = c:MIG
				this = c:TNG
			}
			NOT = { has_variable = dod_treaty_demander }
			NOT = { has_modifier = dod_unequal_treaty }
		}
	}
	when_taken = {
		set_variable = dod_unequal_treaty_sent
		random_country = {
			limit = {
				OR = {
					this = c:CHI
					this = c:MIG
					this = c:TNG
				}
				NOT = { has_variable = dod_treaty_demander }
				NOT = { has_modifier = dod_unequal_treaty }
			}
			set_variable = { name = dod_treaty_demander value = root }
			set_variable = dod_treaty_target
			trigger_event = { id = dod_asi.10 days = 7 }
		}
	}
	ai_chance = { base = 30 }
}

dod_organize_christiania = {
	is_shown = {
		NOT = { exists = c:CHR }

		NOT = { country_has_primary_culture = cu:filipino }
	}
	possible = {
		is_at_war = no
		game_date >= 1870.1.1
	}
	when_taken = {
		dod_release_homeland = { TAG = CHR CULTURE = filipino }

			limit = { exists = c:CHR }

		}
	}
	ai_chance = { base = 20 }
}
