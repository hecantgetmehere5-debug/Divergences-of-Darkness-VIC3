# Gran Colombia, Mexico & the Andes

dod_declare_colombian_independence = {
	is_shown = {
		c:CLM ?= this
		has_journal_entry = je_dod_colombian_independence
		NOT = { has_variable = dod_expel_spanish }
	}
	possible = {
		any_state = {
			owner = {
				OR = {
					c:SPA ?= this
					c:SPC ?= this
				}
			}
			state_region = {
				OR = {
					is_homeland = cu:mexican
					is_homeland = cu:central_american
					is_homeland = cu:granadan
					is_homeland = cu:caribeno
				}
			}
		}
	}
	when_taken = {
		set_variable = dod_expel_spanish
		every_state = {
			limit = {
				owner = {
					OR = {
						c:SPA ?= this
						c:SPC ?= this
					}
				}
				state_region = {
					OR = {
						is_homeland = cu:mexican
						is_homeland = cu:central_american
						is_homeland = cu:granadan
						is_homeland = cu:caribeno
					}
				}
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { value = 100 }
}

dod_colombian_unity_congress = {
	is_shown = {
		OR = {
			c:MEX ?= this
			c:UCA ?= this
			c:GRA ?= this
			c:LST ?= this
		}
		NOT = { exists = c:CLM }
		NOT = { has_global_variable = dod_second_congress }
	}
	possible = {
		is_at_war = no
		game_date >= 1860.1.1
		dod_is_republic = yes
	}
	when_taken = {
		set_global_variable = dod_second_congress
		every_country = {
			limit = {
				OR = {
					c:MEX ?= this
					c:UCA ?= this
					c:GRA ?= this
					c:LST ?= this
				}
				NOT = { this = root }
			}
			save_scope_as = dod_congress_guest
			root = {
				set_variable = { name = dod_unity_invitee value = scope:dod_congress_guest }
			}
			trigger_event = { id = dod_clm.16 days = 30 }
		}
		change_tag = CLM
	}
	ai_chance = { value = 50 }
}
