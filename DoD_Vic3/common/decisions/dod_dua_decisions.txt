# Dual Monarchy: integrating minorities once the Anglois ideal is accepted
# ---------------------------------------------------------------------

dod_integrate_provencal = {
	is_shown = {
		this = c:DUA
		has_variable = dod_anglois_accepted
		NOT = { country_has_primary_culture = cu:provencal }
	}
	possible = {
		NOT = { exists = c:PRV }
		NOT = { exists = c:OCC }
		any_scope_state = { dod_is_provencal_state = yes }
	}
	when_taken = {
		add_primary_culture = cu:provencal
	}
	ai_chance = { base = 50 }
}

# ---------------------------------------------------------------------

dod_reclaim_england = {
	is_shown = {
		this = c:DUA
		exists = c:ENL
		NOT = { has_variable = dod_reclaimed_england }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:great_power
		c:ENL = { country_rank < rank_value:great_power }
	}
	when_taken = {
		set_variable = dod_reclaimed_england
		change_infamy = 20
		every_state = {
			limit = {
				owner = c:ENL
				dod_is_english_state = yes
			}
			state_region = { add_claim = root }
		}
	}
	ai_chance = { base = 50 }
}

# ---------------------------------------------------------------------
# Britain: the Act of Union
# ---------------------------------------------------------------------

dod_act_of_union = {
	is_shown = {
		OR = {
			this = c:ENL
			this = c:SCO
			this = c:IRE
			this = c:WLS
		}
		NOT = { exists = c:ENG }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:great_power
		NOT = {
			any_state = {
				dod_is_british_isles_state = yes
				NOT = { owner = root }
			}
		}
	}
	when_taken = {
		add_primary_culture = cu:english
		add_primary_culture = cu:scottish
		add_primary_culture = cu:irish
		add_primary_culture = cu:welsh
		change_tag = ENG
		add_modifier = { name = dod_act_of_union_mod months = 120 }
	}
	ai_chance = { base = 100 }
}

# ---------------------------------------------------------------------
# The Treaty of Selon
# ---------------------------------------------------------------------

dod_abolish_selon = {
	is_shown = {
		OR = {
			this = c:SPA
			this = c:BRG
			this = c:DUA
		}
		NOT = { has_global_variable = dod_selon_abolished }
	}
	possible = {
		is_at_war = no
		country_rank >= rank_value:great_power
		OR = {
			this = c:SPA
			NOT = { exists = c:SPA }
			c:SPA = { country_rank < rank_value:great_power }
		}
	}
	when_taken = {
		set_global_variable = dod_selon_abolished
		every_country = {
			limit = {
				OR = {
					this = c:SPA
					this = c:BRG
					this = c:DUA
					this = c:SCA
				}
			}
			trigger_event = { id = dod_brg.5 }
		}
	}
	ai_chance = { base = 30 }
}
