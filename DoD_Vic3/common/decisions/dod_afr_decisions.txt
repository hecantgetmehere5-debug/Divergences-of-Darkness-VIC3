# Africa & Oceania

dod_boer_unification = {
	is_shown = {
		country_has_primary_culture = cu:boer
		NOT = { exists = c:BTV }
	}
	possible = {
		is_at_war = no
		any_scope_state = {
			count >= 3
			state_region = {
				is_homeland = cu:boer
			}
		}
	}
	when_taken = {
		every_country = {
			limit = {
				is_subject_of = root
				OR = {
					country_has_primary_culture = cu:boer
				}
			}
			root = { annex = prev }
		}
		change_tag = BTV
	}
	ai_chance = { value = 100 }
}

dod_form_south_africa = {
	is_shown = {
		OR = {

		is_at_war = no
		any_scope_state = {
			count >= 3
			state_region = {
				is_homeland = cu:boer
			}
		}
		any_scope_state = {

		add_primary_culture = cu:boer

	}
	ai_chance = { value = 80 }
}

dod_dahomey_independence = {
	is_shown = {
		this = c:DHY

	when_taken = {

		}

			target_country = scope:dod_dahomey_overlord
			type = dp_independence
		}
	}
	ai_chance = { value = 40 }
}
