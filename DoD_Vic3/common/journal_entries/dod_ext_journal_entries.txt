# Expanded content: Plantagenia, Arcadia, Beornia, Savoy

je_dod_plantagenian_slavery = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	group = je_group_historical_content

	immediate = {
		set_variable = { name = dod_abolition_support value = 10 }
		set_variable = { name = dod_planter_resistance value = 20 }
	}

	current_value = {
		value = root.var:dod_abolition_support
	}
	goal_add_value = {
		add = 100
	}
	progressbar = yes

	on_monthly_pulse = {
		effect = {
			if = {
				limit = { game_date >= 1845.1.1 }
				change_variable = { name = dod_abolition_support add = 1 }
			}
			if = {
				limit = { ig:ig_landowners = { is_in_government = yes } }
				change_variable = { name = dod_planter_resistance add = 1 }
			}
		}
		random_events = {
			80 = 0
			5 = dod_ext.1
			5 = dod_ext.2
			5 = dod_ext.3
			5 = dod_ext.4
		}
	}

	complete = {
		has_law = law_type:law_slavery_banned
	}

	on_complete = {
		trigger_event = { id = dod_ext.5 }
	}

	fail = {
		var:dod_planter_resistance >= 100
	}

	on_fail = {
		trigger_event = { id = dod_ext.6 }
	}

	invalid = {
		NOT = {
			OR = {
				this = c:PLA
				this = c:ARC
			}
		}
	}

	should_be_pinned_by_default = yes
	weight = 1000
}
