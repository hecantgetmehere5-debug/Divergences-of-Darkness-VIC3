# Holy Roman Empire storyline: journal entries
#
# Flow:
#   je_dod_prague_proclamation   -> conditions to issue the Proclamation       (dod_hre.1)
#   je_dod_imperial_diets        -> princes answer, results after 180 days     (dod_hre.20)
#   je_dod_submission_or_conquest -> bring every remaining prince to heel      (dod_hre.40)
#   je_dod_bohemian_question     -> only if the Reich is centralised without the Czechs
#   je_dod_war_for_bohemia       -> after the Bohemian Revolution

je_dod_prague_proclamation = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_historical_content

	# Bohemia plays this storyline and becomes the Holy Roman Empire through the Proclamation
	is_shown_when_inactive = {
		dod_is_emperor = yes
	}

	possible = {
		dod_is_emperor = yes
	}

	# Wiki: the window to form the Empire closes in the late 1840s.



	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 1000

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_dod_prague_proclamation_completed_desc
		color = "gold_positive"
		trigger = {
				is_at_war = no
				country_rank >= rank_value:great_power
				any_country = {
					dod_is_free_prince = yes
				}
	
		}
		effect = {
				trigger_event = { id = dod_hre.1 }
	
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_dod_prague_proclamation_failed_desc
		color = "blue_negative"
		trigger = {
				game_date >= 1849.1.1
	
		}
		effect = {
				trigger_event = { id = dod_rail.11 }
	
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_dod_prague_proclamation_invalid_desc
		trigger = {
				NOT = {
					any_country = {
						dod_is_free_prince = yes
					}
				}
	
		}
	}
}

je_dod_bohemian_question = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	group = je_group_historical_content

	immediate = {
		set_variable = { name = dod_bohemian_unrest value = 10 }
	}

	current_value = {
		value = root.var:dod_bohemian_unrest
	}
	goal_add_value = {
		add = 100
	}
	progressbar = yes

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					OR = {
						has_law = law_type:law_ethnostate
						has_law = law_type:law_national_supremacy
					}
				}
				change_variable = { name = dod_bohemian_unrest add = 2 }
			}
			else_if = {
				limit = { has_law = law_type:law_cultural_exclusion }
				change_variable = { name = dod_bohemian_unrest add = 1 }
			}
		}
		random_events = {
			80 = 0
			20 = dod_hre.51
		}
	}






	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 1000

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_dod_bohemian_question_completed_desc
		color = "gold_positive"
		trigger = {
				OR = {
					country_has_primary_culture = cu:czech
					has_law = law_type:law_multicultural
				}
	
		}
		effect = {
				trigger_event = { id = dod_hre.52 }
	
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_dod_bohemian_question_failed_desc
		color = "blue_negative"
		trigger = {
				var:dod_bohemian_unrest >= 100
	
		}
		effect = {
				trigger_event = { id = dod_hre.54 }
	
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_dod_bohemian_question_invalid_desc
		trigger = {
				NOT = {
					any_scope_state = {
						dod_is_bohemian_state = yes
					}
				}
	
		}
	}
}
